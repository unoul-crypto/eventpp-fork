cmake_minimum_required(VERSION 3.10)
project(eventpp_overhead LANGUAGES CXX)

set(CMAKE_CXX_STANDARD 11)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
find_package(Threads REQUIRED)

if(NOT UPSTREAM_INCLUDE_DIR)
    message(FATAL_ERROR "Set UPSTREAM_INCLUDE_DIR to the baseline's include directory")
endif()
get_filename_component(FORK_INCLUDE_DIR "../../../include" ABSOLUTE BASE_DIR "${CMAKE_CURRENT_SOURCE_DIR}")

foreach(variant upstream fork)
    foreach(mode timing allocations)
        set(target "overhead_${variant}_${mode}")
        add_executable(${target} overhead.cpp)
        target_link_libraries(${target} PRIVATE Threads::Threads)
        if(variant STREQUAL "fork")
            target_include_directories(${target} PRIVATE "${FORK_INCLUDE_DIR}")
            target_compile_definitions(${target} PRIVATE OVERHEAD_FORK=1)
        else()
            target_include_directories(${target} PRIVATE "${UPSTREAM_INCLUDE_DIR}")
        endif()
        if(mode STREQUAL "allocations")
            target_compile_definitions(${target} PRIVATE OVERHEAD_ALLOCATIONS=1)
        endif()
        if(MSVC)
            target_compile_options(${target} PRIVATE /W4)
        else()
            target_compile_options(${target} PRIVATE -Wall -Wextra -Wpedantic)
        endif()
    endforeach()
endforeach()
